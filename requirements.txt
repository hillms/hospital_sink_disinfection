# =============================================================================
# Hospital Sink Disinfection Study - R Package Requirements
#
# Install these packages before running the analysis pipeline
# =============================================================================

# Core data manipulation packages
install.packages(c(
  "dplyr",         # Data manipulation and transformation
  "tidyr",         # Data tidying and reshaping
  "readr",         # Fast data reading
  "tibble",        # Modern data frames
  "magrittr"       # Pipe operators
))

# Statistical analysis packages
install.packages(c(
  "vegan",         # Community ecology analysis and ordination
  "lme4",          # Linear mixed-effects models
  "broom",         # Tidy statistical outputs
  "car",           # Companion to Applied Regression
  "emmeans"        # Estimated marginal means
))

# Compositional data analysis
if (!require("BiocManager", quietly = TRUE)) {
  install.packages("BiocManager")
}
BiocManager::install(c(
  "ALDEx2",        # Differential abundance analysis for compositional data
  "decontam",      # Contaminant identification (preprocessing/02_decontam.R)
  "DESeq2",        # Calibration comparison (analysis/04b_da_calibration.R)
  "edgeR"          # Calibration comparison (analysis/04b_da_calibration.R)
))
install.packages("data.table")  # Fast reading of the feature tables

# Specialized ecological packages
install.packages(c(
  "indicspecies",  # Indicator species analysis
  "igraph",        # Network analysis and visualization
  "corrplot"       # Correlation matrix visualization
))

# Data visualization packages
install.packages(c(
  "ggplot2",       # Grammar of graphics
  "ggrepel",       # Text repelling for clean plots
  "patchwork",     # Combining multiple plots
  "cowplot",       # Publication-ready plot themes
  "gridExtra",     # Grid-based plot arrangements
  "scales",        # Scale functions for ggplot2
  "ggpubr"         # Publication-ready plots
))

# Color packages
install.packages(c(
  "RColorBrewer",  # ColorBrewer palettes
  "viridis",       # Perceptually uniform color scales
  "wesanderson"    # Wes Anderson color palettes
))

# Additional utility packages
install.packages(c(
  "here",          # Project-relative file paths
  "lubridate",     # Date and time manipulation
  "stringr",       # String manipulation
  "forcats"        # Factor manipulation
))
